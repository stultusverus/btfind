# btfind Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a BitTorrent DHT sniffer in Rust that crawls the DHT, fetches torrent metadata, stores results in SQLite, and provides CLI search.

**Architecture:** Tokio async runtime drives three subsystems: a DHT crawler over UDP (KRPC/bencode), a metadata fetcher over TCP (BitTorrent wire protocol + BEP 9), and a SQLite store accessed via blocking threadpool. A `clap` CLI orchestrates them with `run`, `search`, `stats`, `prune` subcommands.

**Tech Stack:** Rust, tokio, serde_bencode, rusqlite, clap, sha1, rand, tracing, toml, dirs, chrono

**File Structure:**
```
Cargo.toml
src/
├── main.rs          — CLI entry point, subcommand dispatch
├── config.rs        — Config struct, TOML loading, defaults
├── types.rs         — NodeId, InfoHash, PeerContact, NodeContact
├── bencode.rs       — KRPC message types, serialize/deserialize with serde_bencode
├── routing.rs       — Kademlia bucket-based routing table
├── dht.rs           — DHT crawler: bootstrap, find_node, get_peers, discovery loop
├── wire.rs          — BitTorrent wire protocol messages (handshake, extensions, ut_metadata)
├── metadata.rs      — Metadata fetcher: connect to peers, request/fetch torrent metadata
├── store.rs         — SQLite operations: init schema, insert/search/stats/prune
```

```rust
pub fn random_node_id() -> NodeId {
    use rand::Rng;
    let mut rng = rand::thread_rng();
    let mut id = [0u8; NODE_ID_LEN];
    rng.fill(&mut id);
    id
}

pub fn dummy_node_id() -> NodeId {
    random_node_id()
}

pub fn info_hash_from_hex(hex: &str) -> Option<InfoHash> {
    let bytes = hex::decode(hex).ok()?;
    if bytes.len() != INFO_HASH_LEN {
        return None;
    }
    let mut hash = [0u8; INFO_HASH_LEN];
    hash.copy_from_slice(&bytes);
    Some(hash)
}

pub fn info_hash_to_hex(hash: &InfoHash) -> String {
    hex::encode(hash)
}

pub fn node_id_distance(a: &NodeId, b: &NodeId) -> [u8; NODE_ID_LEN] {
    let mut dist = [0u8; NODE_ID_LEN];
    for i in 0..NODE_ID_LEN {
        dist[i] = a[i] ^ b[i];
    }
    dist
}
```

```rust
use std::collections::BTreeMap;
use serde::{Deserialize, Serialize};

/// A value in a bencoded dictionary.
#[derive(Debug, Clone, PartialEq)]
pub enum KrpcValue {
    Bytes(Vec<u8>),
    Int(i64),
    List(Vec<KrpcValue>),
    Dict(BTreeMap<String, KrpcValue>),
}

/// A parsed KRPC message.
#[derive(Debug, Clone)]
pub enum KrpcMessage {
    Query {
        t: Vec<u8>,
        y: String,
        q: String,
        a: BTreeMap<String, KrpcValue>,
    },
    Response {
        t: Vec<u8>,
        y: String,
        r: BTreeMap<String, KrpcValue>,
    },
    Error {
        t: Vec<u8>,
        y: String,
        e: KrpcError,
    },
}

#[derive(Debug, Clone)]
pub struct KrpcError {
    pub code: i64,
    pub description: String,
}

/// Convert our KrpcValue to serde_bencode::value::Value for serialization.
fn krpc_to_bencode(v: &KrpcValue) -> serde_bencode::value::Value {
    match v {
        KrpcValue::Bytes(b) => serde_bencode::value::Value::Bytes(b.clone()),
        KrpcValue::Int(i) => serde_bencode::value::Value::Int(*i),
        KrpcValue::List(l) => {
            serde_bencode::value::Value::List(l.iter().map(krpc_to_bencode).collect())
        }
        KrpcValue::Dict(d) => {
            let mut map = BTreeMap::new();
            for (k, val) in d {
                map.insert(k.clone().into_bytes(), krpc_to_bencode(val));
            }
            serde_bencode::value::Value::Dict(map)
        }
    }
}

/// Convert serde_bencode::value::Value back to our KrpcValue.
fn bencode_to_krpc(v: &serde_bencode::value::Value) -> KrpcValue {
    match v {
        serde_bencode::value::Value::Bytes(b) => KrpcValue::Bytes(b.clone()),
        serde_bencode::value::Value::Int(i) => KrpcValue::Int(*i),
        serde_bencode::value::Value::List(l) => {
            KrpcValue::List(l.iter().map(bencode_to_krpc).collect())
        }
        serde_bencode::value::Value::Dict(d) => {
            let mut map = BTreeMap::new();
            for (k, val) in d {
                if let Ok(key) = String::from_utf8(k.clone()) {
                    map.insert(key, bencode_to_krpc(val));
                }
            }
            KrpcValue::Dict(map)
        }
    }
}

impl KrpcMessage {
    pub fn to_bytes(&self) -> Vec<u8> {
        match self {
            KrpcMessage::Query { t, y, q, a } => {
                let mut dict = BTreeMap::new();
                dict.insert("t".as_bytes().to_vec(), serde_bencode::value::Value::Bytes(t.clone()));
                dict.insert("y".as_bytes().to_vec(), serde_bencode::value::Value::Bytes(y.as_bytes().to_vec()));
                dict.insert("q".as_bytes().to_vec(), serde_bencode::value::Value::Bytes(q.as_bytes().to_vec()));
                let mut a_dict = BTreeMap::new();
                for (k, v) in a {
                    a_dict.insert(k.as_bytes().to_vec(), krpc_to_bencode(v));
                }
                dict.insert("a".as_bytes().to_vec(), serde_bencode::value::Value::Dict(a_dict));
                serde_bencode::to_bytes(&serde_bencode::value::Value::Dict(dict)).unwrap_or_default()
            }
            KrpcMessage::Response { t, y, r } => {
                let mut dict = BTreeMap::new();
                dict.insert("t".as_bytes().to_vec(), serde_bencode::value::Value::Bytes(t.clone()));
                dict.insert("y".as_bytes().to_vec(), serde_bencode::value::Value::Bytes(y.as_bytes().to_vec()));
                let mut r_dict = BTreeMap::new();
                for (k, v) in r {
                    r_dict.insert(k.as_bytes().to_vec(), krpc_to_bencode(v));
                }
                dict.insert("r".as_bytes().to_vec(), serde_bencode::value::Value::Dict(r_dict));
                serde_bencode::to_bytes(&serde_bencode::value::Value::Dict(dict)).unwrap_or_default()
            }
            KrpcMessage::Error { t, y, e } => {
                let mut dict = BTreeMap::new();
                dict.insert("t".as_bytes().to_vec(), serde_bencode::value::Value::Bytes(t.clone()));
                dict.insert("y".as_bytes().to_vec(), serde_bencode::value::Value::Bytes(y.as_bytes().to_vec()));
                let mut e_list = Vec::new();
                e_list.push(serde_bencode::value::Value::Int(e.code));
                e_list.push(serde_bencode::value::Value::Bytes(e.description.as_bytes().to_vec()));
                dict.insert("e".as_bytes().to_vec(), serde_bencode::value::Value::List(e_list));
                serde_bencode::to_bytes(&serde_bencode::value::Value::Dict(dict)).unwrap_or_default()
            }
        }
    }

    pub fn from_bytes(data: &[u8]) -> Result<Self, String> {
        let value: serde_bencode::value::Value = serde_bencode::from_bytes(data)
            .map_err(|e| format!("bencode decode error: {}", e))?;

        let dict = match &value {
            serde_bencode::value::Value::Dict(d) => d,
            _ => return Err("top-level value is not a dict".into()),
        };

        let t = get_bytes(dict, "t").ok_or("missing t")?;
        let y = get_string(dict, "y").ok_or("missing y")?;

        match y.as_str() {
            "q" => {
                let q = get_string(dict, "q").ok_or("missing q")?;
                let a = get_dict(dict, "a").unwrap_or_default();
                let mut a_map = BTreeMap::new();
                for (k, v) in a {
                    if let Ok(key) = String::from_utf8(k.clone()) {
                        a_map.insert(key, bencode_to_krpc(v));
                    }
                }
                Ok(KrpcMessage::Query { t, y, q, a: a_map })
            }
            "r" => {
                let r = get_dict(dict, "r").unwrap_or_default();
                let mut r_map = BTreeMap::new();
                for (k, v) in r {
                    if let Ok(key) = String::from_utf8(k.clone()) {
                        r_map.insert(key, bencode_to_krpc(v));
                    }
                }
                Ok(KrpcMessage::Response { t, y, r: r_map })
            }
            "e" => {
                let e_list = get_list(dict, "e").ok_or("missing e")?;
                let code = match e_list.first() {
                    Some(serde_bencode::value::Value::Int(i)) => *i,
                    _ => return Err("invalid error code".into()),
                };
                let description = match e_list.get(1) {
                    Some(serde_bencode::value::Value::Bytes(b)) => {
                        String::from_utf8_lossy(b).to_string()
                    }
                    _ => "unknown error".to_string(),
                };
                Ok(KrpcMessage::Error {
                    t,
                    y,
                    e: KrpcError { code, description },
                })
            }
            _ => Err(format!("unknown message type: {}", y)),
        }
    }

    pub fn is_query(&self) -> bool {
        matches!(self, KrpcMessage::Query { .. })
    }

    pub fn is_response(&self) -> bool {
        matches!(self, KrpcMessage::Response { .. })
    }

    pub fn transaction_id(&self) -> &[u8] {
        match self {
            KrpcMessage::Query { t, .. } | KrpcMessage::Response { t, .. } | KrpcMessage::Error { t, .. } => t,
        }
    }
}

fn get_bytes(dict: &BTreeMap<Vec<u8>, serde_bencode::value::Value>, key: &str) -> Option<Vec<u8>> {
    match dict.get(&key.as_bytes().to_vec())? {
        serde_bencode::value::Value::Bytes(b) => Some(b.clone()),
        _ => None,
    }
}

fn get_string(dict: &BTreeMap<Vec<u8>, serde_bencode::value::Value>, key: &str) -> Option<String> {
    get_bytes(dict, key).map(|b| String::from_utf8_lossy(&b).to_string())
}

fn get_dict(dict: &BTreeMap<Vec<u8>, serde_bencode::value::Value>, key: &str) -> Option<BTreeMap<Vec<u8>, serde_bencode::value::Value>> {
    match dict.get(&key.as_bytes().to_vec())? {
        serde_bencode::value::Value::Dict(d) => Some(d.clone()),
        _ => None,
    }
}

fn get_list(dict: &BTreeMap<Vec<u8>, serde_bencode::value::Value>, key: &str) -> Option<Vec<serde_bencode::value::Value>> {
    match dict.get(&key.as_bytes().to_vec())? {
        serde_bencode::value::Value::List(l) => Some(l.clone()),
        _ => None,
    }
}
```

```rust
use crate::types::{NodeId, NodeContact, node_id_distance, NODE_ID_LEN};
use std::collections::BTreeMap;
use std::net::SocketAddrV4;
use std::time::{Duration, Instant};

const BUCKET_COUNT: usize = NODE_ID_LEN * 8;
const MAX_NODES_PER_BUCKET: usize = 8;

pub struct Bucket {
    nodes: BTreeMap<NodeId, NodeContact>,
}

impl Bucket {
    fn new() -> Self {
        Bucket {
            nodes: BTreeMap::new(),
        }
    }

    fn len(&self) -> usize {
        self.nodes.len()
    }
}

pub struct RoutingTable {
    our_id: NodeId,
    buckets: Vec<Bucket>,
}

impl RoutingTable {
    pub fn new(our_id: NodeId) -> Self {
        let mut buckets = Vec::with_capacity(BUCKET_COUNT);
        for _ in 0..BUCKET_COUNT {
            buckets.push(Bucket::new());
        }
        RoutingTable { our_id, buckets }
    }

    fn bucket_index(&self, node_id: &NodeId) -> usize {
        let dist = node_id_distance(&self.our_id, node_id);
        // Find the index of the first non-zero byte, then the first set bit within it
        for (byte_idx, &byte) in dist.iter().enumerate() {
            if byte != 0 {
                let bit = byte.leading_zeros() as usize;
                return byte_idx * 8 + bit;
            }
        }
        // Node ID equals our ID - put in last bucket
        BUCKET_COUNT - 1
    }

    pub fn add_node(&mut self, id: NodeId, addr: SocketAddrV4) -> bool {
        // Don't add ourselves
        if id == self.our_id {
            return false;
        }

        let bucket_idx = self.bucket_index(&id);
        let bucket = &mut self.buckets[bucket_idx];

        if bucket.nodes.contains_key(&id) {
            bucket.nodes.get_mut(&id).unwrap().last_seen = Instant::now();
            return false; // already present
        }

        if bucket.nodes.len() >= MAX_NODES_PER_BUCKET {
            // If bucket is full, we should ping the oldest node to see if it's alive.
            // For simplicity, just return false indicating we couldn't add.
            return false;
        }

        bucket.nodes.insert(
            id,
            NodeContact {
                id,
                addr,
                last_seen: Instant::now(),
            },
        );
        true
    }

    pub fn closest_nodes(&self, target: &NodeId, count: usize) -> Vec<NodeContact> {
        let mut all: Vec<&NodeContact> = self
            .buckets
            .iter()
            .flat_map(|b| b.nodes.values())
            .collect();

        all.sort_by_key(|node| node_id_distance(&node.id, target));
        all.truncate(count);

        all.into_iter().cloned().collect()
    }

    pub fn remove_stale_nodes(&mut self, max_age: Duration) -> usize {
        let mut removed = 0;
        let now = Instant::now();
        for bucket in &mut self.buckets {
            let stale: Vec<NodeId> = bucket
                .nodes
                .iter()
                .filter(|(_, c)| now.duration_since(c.last_seen) > max_age)
                .map(|(id, _)| *id)
                .collect();
            for id in stale {
                bucket.nodes.remove(&id);
                removed += 1;
            }
        }
        removed
    }

    pub fn node_count(&self) -> usize {
        self.buckets.iter().map(|b| b.nodes.len()).sum()
    }

    pub fn random_nodes(&self, count: usize) -> Vec<NodeContact> {
        let all: Vec<&NodeContact> = self
            .buckets
            .iter()
            .flat_map(|b| b.nodes.values())
            .collect();

        if all.is_empty() {
            return vec![];
        }

        use rand::seq::SliceRandom;
        let mut rng = rand::thread_rng();
        all.choose_multiple(&mut rng, count.min(all.len()))
            .cloned()
            .collect()
    }

    pub fn update_last_seen(&mut self, node_id: &NodeId) {
        let bucket_idx = self.bucket_index(node_id);
        if let Some(node) = self.buckets[bucket_idx].nodes.get_mut(node_id) {
            node.last_seen = Instant::now();
        }
    }
}
```

```rust
}
```

```rust
pub struct CrawlStats {

}

pub struct Store {
    conn: Mutex<Connection>,
}

impl Store {
    pub fn open(path: &Path) -> Result<Self, rusqlite::Error> {
        let conn = Connection::open(path)?;
        let store = Store {
            conn: Mutex::new(conn),
        };
        store.init_schema()?;
        Ok(store)
    }

    pub fn open_in_memory() -> Result<Self, rusqlite::Error> {
        let conn = Connection::open_in_memory()?;
        let store = Store {
            conn: Mutex::new(conn),
        };
        store.init_schema()?;
        Ok(store)
    }

    fn init_schema(&self) -> Result<(), rusqlite::Error> {
        let conn = self.conn.lock().unwrap();
        conn.execute_batch(
            "
            CREATE TABLE IF NOT EXISTS torrents (
                info_hash         TEXT PRIMARY KEY,
                name              TEXT,
                piece_length      INTEGER,
                total_size        INTEGER NOT NULL DEFAULT 0,
                file_count        INTEGER NOT NULL DEFAULT 0,
                source            TEXT NOT NULL DEFAULT 'unknown',
                first_seen        INTEGER NOT NULL,
                last_seen         INTEGER NOT NULL,
                metadata_complete INTEGER NOT NULL DEFAULT 0
            );

            CREATE TABLE IF NOT EXISTS files (
                id          INTEGER PRIMARY KEY AUTOINCREMENT,
                torrent_id  TEXT NOT NULL REFERENCES torrents(info_hash) ON DELETE CASCADE,
                path        TEXT NOT NULL,
                size        INTEGER NOT NULL
            );

            CREATE INDEX IF NOT EXISTS idx_torrents_first_seen ON torrents(first_seen);
            CREATE INDEX IF NOT EXISTS idx_torrents_last_seen ON torrents(last_seen);
            CREATE INDEX IF NOT EXISTS idx_torrents_name ON torrents(name);
            CREATE INDEX IF NOT EXISTS idx_files_torrent ON files(torrent_id);

            PRAGMA journal_mode=WAL;
            PRAGMA foreign_keys=ON;
            ",
        )?;
        Ok(())
    }

    /// Insert or update a torrent record. Returns the rowid.
    pub fn upsert_torrent(
        &self,
        info_hash: InfoHash,
        name: Option<String>,
        total_size: i64,
        file_count: i64,
        source: &str,
    ) -> Result<i64, rusqlite::Error> {
        let conn = self.conn.lock().unwrap();
        let hash_hex = hex::encode(info_hash);
        let now = chrono::Utc::now().timestamp();

        // If there's an existing record, preserve first_seen
        let existing_first_seen: Option<i64> = conn
            .query_row(
                "SELECT first_seen FROM torrents WHERE info_hash = ?1",
                params![hash_hex],
                |row| row.get(0),
            )
            .ok();

        let first_seen = existing_first_seen.unwrap_or(now);

        conn.execute(
            "INSERT OR REPLACE INTO torrents (info_hash, name, total_size, file_count, source, first_seen, last_seen, metadata_complete)
             VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8)",
            params![
                hash_hex,
                name,
                total_size,
                file_count,
                source,
                first_seen,
                now,
                if name.is_some() { 1i32 } else { 0i32 },
            ],
        )?;
        Ok(conn.last_insert_rowid())
    }

    /// Mark a torrent's metadata as complete (set if name is populated).
    pub fn mark_metadata_complete(
        &self,
        info_hash: &InfoHash,
        name: &str,
        piece_length: i64,
        total_size: i64,
        file_count: i64,
    ) -> Result<(), rusqlite::Error> {
        let conn = self.conn.lock().unwrap();
        let hash_hex = hex::encode(info_hash);
        conn.execute(
            "UPDATE torrents SET name = ?2, piece_length = ?3, total_size = ?4, file_count = ?5, metadata_complete = 1, last_seen = unixepoch()
             WHERE info_hash = ?1",
            params![hash_hex, name, piece_length, total_size, file_count],
        )?;
        Ok(())
    }

    /// Insert files for a torrent. Replaces existing files.
    pub fn insert_files(
        &self,
        info_hash: &InfoHash,
        files: &[(String, i64)],
    ) -> Result<(), rusqlite::Error> {
        let conn = self.conn.lock().unwrap();
        let hash_hex = hex::encode(info_hash);

        conn.execute("DELETE FROM files WHERE torrent_id = ?1", params![hash_hex])?;

        let mut stmt = conn.prepare(
            "INSERT INTO files (torrent_id, path, size) VALUES (?1, ?2, ?3)",
        )?;

        for (path, size) in files {
            stmt.execute(params![hash_hex, path, size])?;
        }

        Ok(())
    }

    pub fn get_torrent(&self, info_hash: &InfoHash) -> Result<Option<TorrentRecord>, rusqlite::Error> {
        let conn = self.conn.lock().unwrap();
        let hash_hex = hex::encode(info_hash);

        let mut stmt = conn.prepare(
            "SELECT info_hash, name, piece_length, total_size, file_count, source, first_seen, last_seen, metadata_complete
             FROM torrents WHERE info_hash = ?1",
        )?;

        let mut rows = stmt.query_map(params![hash_hex], row_to_torrent)?;
        match rows.next() {
            Some(Ok(record)) => Ok(Some(record)),
            Some(Err(e)) => Err(e),
            None => Ok(None),
        }
    }

    pub fn search_torrents(
        &self,
        query: &str,
        limit: usize,
    ) -> Result<Vec<TorrentRecord>, rusqlite::Error> {
        let conn = self.conn.lock().unwrap();
        let pattern = format!("%{}%", query);

        let mut stmt = conn.prepare(
            "SELECT info_hash, name, piece_length, total_size, file_count, source, first_seen, last_seen, metadata_complete
             FROM torrents
             WHERE name LIKE ?1
             ORDER BY last_seen DESC
             LIMIT ?2",
        )?;

        let rows = stmt.query_map(params![pattern, limit as i64], row_to_torrent)?;
        let mut results = Vec::new();
        for row in rows {
            results.push(row?);
        }
        Ok(results)
    }

    pub fn get_stats(&self) -> Result<CrawlStats, rusqlite::Error> {
        let conn = self.conn.lock().unwrap();
        conn.query_row(
            "SELECT
                COUNT(*) as total,
                COALESCE(SUM(CASE WHEN metadata_complete = 1 THEN 1 ELSE 0 END), 0) as complete,
                COALESCE((SELECT COUNT(*) FROM files), 0) as files_total,
                COALESCE(SUM(total_size), 0) as total_size
             FROM torrents",
            [],
            |row| {
                Ok(CrawlStats {
                    total_torrents: row.get(0)?,
                    total_metadata_complete: row.get(1)?,
                    total_files: row.get(2)?,
                    total_size: row.get(3)?,
                })
            },
        )
    }

    pub fn prune_old_torrents(&self, days: i64) -> Result<i64, rusqlite::Error> {
        let conn = self.conn.lock().unwrap();
        let cutoff = chrono::Utc::now().timestamp() - days * 86400;
        let removed = conn.execute(
            "DELETE FROM torrents WHERE last_seen < ?1",
            params![cutoff],
        )?;
        Ok(removed as i64)
    }
}

fn row_to_torrent(row: &rusqlite::Row) -> rusqlite::Result<TorrentRecord> {
    let hash_hex: String = row.get(0)?;
    let hash_bytes = hex::decode(&hash_hex).unwrap_or_default();
    let mut info_hash: InfoHash = [0u8; 20];
    let len = hash_bytes.len().min(20);
    info_hash[..len].copy_from_slice(&hash_bytes[..len]);

    Ok(TorrentRecord {
        info_hash,
        name: row.get(1)?,
        piece_length: row.get(2)?,
        total_size: row.get(3)?,
        file_count: row.get(4)?,
        source: row.get(5)?,
        first_seen: row.get(6)?,
        last_seen: row.get(7)?,
        metadata_complete: row.get::<_, i32>(8)? != 0,
    })
}
```

```rust
use crate::bencode::{KrpcMessage, KrpcValue};
use crate::routing::RoutingTable;
use crate::types;
use crate::types::{InfoHash, NodeContact, NodeId, PeerContact, random_node_id, NODE_ID_LEN, TRANSACTION_ID_LEN};
use std::collections::{BTreeMap, HashMap};
use std::net::{SocketAddrV4, UdpSocket};
use std::sync::atomic::{AtomicU16, Ordering};
use std::sync::Arc;
use tokio::net::UdpSocket as TokioUdpSocket;
use tokio::sync::mpsc;
use tokio::time::{self, Duration};

static NEXT_TX: AtomicU16 = AtomicU16::new(1);

fn next_transaction_id() -> u16 {
    NEXT_TX.fetch_add(1, Ordering::SeqCst)
}

fn transaction_id_bytes(id: u16) -> [u8; TRANSACTION_ID_LEN] {
    id.to_be_bytes()
}

/// Build a "ping" query.
pub fn build_ping_query(node_id: &NodeId, tx_id: u16) -> KrpcMessage {
    let mut a = BTreeMap::new();
    a.insert("id".to_string(), KrpcValue::Bytes(node_id.to_vec()));

    KrpcMessage::Query {
        t: transaction_id_bytes(tx_id).to_vec(),
        y: "q".to_string(),
        q: "ping".to_string(),
        a,
    }
}

/// Build a "find_node" query.
pub fn build_find_node_query(node_id: &NodeId, target: &NodeId, tx_id: u16) -> KrpcMessage {
    let mut a = BTreeMap::new();
    a.insert("id".to_string(), KrpcValue::Bytes(node_id.to_vec()));
    a.insert("target".to_string(), KrpcValue::Bytes(target.to_vec()));

    KrpcMessage::Query {
        t: transaction_id_bytes(tx_id).to_vec(),
        y: "q".to_string(),
        q: "find_node".to_string(),
        a,
    }
}

/// Build a "get_peers" query.
pub fn build_get_peers_query(node_id: &NodeId, info_hash: &InfoHash, tx_id: u16) -> KrpcMessage {
    let mut a = BTreeMap::new();
    a.insert("id".to_string(), KrpcValue::Bytes(node_id.to_vec()));
    a.insert("info_hash".to_string(), KrpcValue::Bytes(info_hash.to_vec()));

    KrpcMessage::Query {
        t: transaction_id_bytes(tx_id).to_vec(),
        y: "q".to_string(),
        q: "get_peers".to_string(),
        a,
    }
}

/// Build a "find_node" response with compact node info.
pub fn build_find_node_response(node_id: &NodeId, nodes: &[u8], tx: Vec<u8>) -> KrpcMessage {
    let mut r = BTreeMap::new();
    r.insert("id".to_string(), KrpcValue::Bytes(node_id.to_vec()));
    r.insert("nodes".to_string(), KrpcValue::Bytes(nodes.to_vec()));

    KrpcMessage::Response {
        t: tx,
        y: "r".to_string(),
        r,
    }
}

/// Decode compact node info from a "nodes" response. Each entry is 26 bytes: 20-byte ID + 4-byte IP + 2-byte port.
pub fn decode_compact_nodes(data: &[u8]) -> Vec<NodeContact> {
    let mut contacts = Vec::new();
    let entry_size = NODE_ID_LEN + 6; // 20 bytes ID + 4 bytes IP + 2 bytes port

    for chunk in data.chunks(entry_size) {
        if chunk.len() < entry_size {
            break;
        }
        let mut id: NodeId = [0u8; NODE_ID_LEN];
        id.copy_from_slice(&chunk[..NODE_ID_LEN]);

        let ip = std::net::Ipv4Addr::new(chunk[20], chunk[21], chunk[22], chunk[23]);
        let port = u16::from_be_bytes([chunk[24], chunk[25]]);
        let addr = SocketAddrV4::new(ip, port);

        contacts.push(NodeContact {
            id,
            addr,
            last_seen: std::time::Instant::now(),
        });
    }
    contacts
}

/// Encode node contacts into compact format for find_node responses.
pub fn encode_compact_nodes(nodes: &[NodeContact]) -> Vec<u8> {
    let mut data = Vec::with_capacity(nodes.len() * 26);
    for node in nodes {
        data.extend_from_slice(&node.id);
        data.extend_from_slice(&node.addr.ip().octets());
        data.extend_from_slice(&node.addr.port().to_be_bytes());
    }
    data
}

/// Parse a get_peers response. Returns (peers, nodes) depending on whether
/// the response contains "values" (compact peer info) or "nodes" (compact node info).
pub fn parse_get_peers_response(msg: &KrpcMessage) -> (Vec<PeerContact>, Vec<NodeContact>) {
    match msg {
        KrpcMessage::Response { r, .. } => {
            let peers = if let Some(KrpcValue::List(values)) = r.get("values") {
                decode_compact_peer_values(values)
            } else if let Some(KrpcValue::Bytes(data)) = r.get("values") {
                // Some implementations return values as a single byte string
                decode_compact_peers(data)
            } else {
                vec![]
            };

            let nodes = if let Some(KrpcValue::Bytes(data)) = r.get("nodes") {
                decode_compact_nodes(data)
            } else {
                vec![]
            };

            (peers, nodes)
        }
        _ => (vec![], vec![]),
    }
}

fn decode_compact_peer_values(values: &[KrpcValue]) -> Vec<PeerContact> {
    let mut peers = Vec::new();
    for val in values {
        if let KrpcValue::Bytes(data) = val {
            peers.extend(decode_compact_peers(data));
        }
    }
    peers
}

fn decode_compact_peers(data: &[u8]) -> Vec<PeerContact> {
    let mut peers = Vec::new();
    for chunk in data.chunks(6) {
        if chunk.len() < 6 {
            break;
        }
        let ip = std::net::Ipv4Addr::new(chunk[0], chunk[1], chunk[2], chunk[3]);
        let port = u16::from_be_bytes([chunk[4], chunk[5]]);
        peers.push(PeerContact {
            addr: SocketAddrV4::new(ip, port),
        });
    }
    peers
}

/// Extract the node ID from a KRPC response/message's "id" field.
pub fn extract_node_id(msg: &KrpcMessage) -> Option<NodeId> {
    let r = match msg {
        KrpcMessage::Response { r, .. } => Some(r),
        KrpcMessage::Query { a, .. } => Some(a),
        _ => None,
    }?;

    match r.get("id")? {
        KrpcValue::Bytes(b) if b.len() == NODE_ID_LEN => {
            let mut id: NodeId = [0u8; NODE_ID_LEN];
            id.copy_from_slice(b);
            Some(id)
        }
        _ => None,
    }
}

/// The DHT crawler state machine.
pub struct DhtCrawler {
    pub our_id: NodeId,
    pub socket: Arc<TokioUdpSocket>,
    pub routing: RoutingTable,
    pub info_hash_tx: mpsc::UnboundedSender<InfoHash>,
}

impl DhtCrawler {
    pub fn new(
        socket: Arc<TokioUdpSocket>,
        info_hash_tx: mpsc::UnboundedSender<InfoHash>,
    ) -> Self {
        let our_id = random_node_id();
        DhtCrawler {
            our_id,
            socket,
            routing: RoutingTable::new(our_id),
            info_hash_tx,
        }
    }

    /// Bootstrap by sending find_node to hardcoded bootstrap nodes.
    pub async fn bootstrap(&mut self) {
        let bootstrap_nodes: Vec<SocketAddrV4> = vec![
            "67.215.246.10:6881".parse().unwrap(),      // router.bittorrent.com
            "87.98.162.88:6881".parse().unwrap(),        // dht.transmissionbt.com
            "82.221.103.244:6881".parse().unwrap(),       // router.utorrent.com
        ];

        for addr in &bootstrap_nodes {
            let target = random_node_id();
            let msg = build_find_node_query(&self.our_id, &target, next_transaction_id());
            let data = msg.to_bytes();
            if let Err(e) = self.socket.send_to(&data, *addr).await {
                tracing::warn!("bootstrap send error to {}: {}", addr, e);
            }
        }
    }

    /// Main crawl loop: read responses, send queries.
    pub async fn run(
        mut self,
        get_peers_interval: Duration,
        bucket_refresh: Duration,
        continuous: bool,
    ) {
        let mut buf = vec![0u8; 8192];
        let mut peer_discover_tick = time::interval(get_peers_interval);
        let mut refresh_tick = time::interval(bucket_refresh);

        // Start by bootstrapping
        self.bootstrap().await;

        // The first tick fires immediately, we want to delay it
        peer_discover_tick.tick().await; // consume first immediate tick
        refresh_tick.tick().await;

        loop {
            tokio::select! {
                result = self.socket.recv_from(&mut buf) => {
                    match result {
                        Ok((len, src)) => {
                            self.handle_packet(&buf[..len], src).await;
                        }
                        Err(e) => {
                            tracing::error!("socket recv error: {}", e);
                        }
                    }
                }

                _ = peer_discover_tick.tick() => {
                    self.discover_peers().await;
                }

                _ = refresh_tick.tick() => {
                    let removed = self.routing.remove_stale_nodes(Duration::from_secs(900));
                    if removed > 0 {
                        tracing::debug!("removed {} stale nodes", removed);
                    }
                }
            }
        }
    }

    async fn handle_packet(&mut self, data: &[u8], src: SocketAddrV4) {
        let msg = match KrpcMessage::from_bytes(data) {
            Ok(m) => m,
            Err(e) => {
                tracing::debug!("bad krpc from {}: {}", src, e);
                return;
            }
        };

        // Update routing table with sender's node ID
        if let Some(node_id) = extract_node_id(&msg) {
            self.routing.add_node(node_id, src);
            self.routing.update_last_seen(&node_id);
        }

        match &msg {
            KrpcMessage::Query { q, a, t, .. } => {
                match q.as_str() {
                    "ping" => {
                        // Respond with our ID
                        let mut r = BTreeMap::new();
                        r.insert("id".to_string(), KrpcValue::Bytes(self.our_id.to_vec()));
                        let resp = KrpcMessage::Response {
                            t: t.clone(),
                            y: "r".to_string(),
                            r,
                        };
                        let _ = self.socket.send_to(&resp.to_bytes(), src).await;
                    }
                    "get_peers" => {
                        // Respond with closest nodes (we don't have peers to advertise)
                        if let Some(KrpcValue::Bytes(info_hash)) = a.get("info_hash") {
                            let mut info_hash_arr: InfoHash = [0u8; 20];
                            let len = info_hash.len().min(20);
                            info_hash_arr[..len].copy_from_slice(&info_hash[..len]);
                            // Publish the info_hash we were queried about
                            let _ = self.info_hash_tx.send(info_hash_arr);
                        }
                        // Send closest nodes
                        let target = random_node_id();
                        let closest = self.routing.closest_nodes(&target, 8);
                        let nodes = encode_compact_nodes(&closest);
                        let mut r = BTreeMap::new();
                        r.insert("id".to_string(), KrpcValue::Bytes(self.our_id.to_vec()));
                        r.insert("nodes".to_string(), KrpcValue::Bytes(nodes));
                        r.insert("token".to_string(), KrpcValue::Bytes(b"btfind".to_vec()));
                        let resp = KrpcMessage::Response {
                            t: t.clone(),
                            y: "r".to_string(),
                            r,
                        };
                        let _ = self.socket.send_to(&resp.to_bytes(), src).await;
                    }
                    "find_node" => {
                        let closest = self.routing.closest_nodes(&random_node_id(), 8);
                        let nodes = encode_compact_nodes(&closest);
                        let resp = build_find_node_response(&self.our_id, &nodes, t.clone());
                        let _ = self.socket.send_to(&resp.to_bytes(), src).await;
                    }
                    "announce_peer" => {
                        // Another node is announcing it has a torrent
                        if let Some(KrpcValue::Bytes(info_hash)) = a.get("info_hash") {
                            let mut info_hash_arr: InfoHash = [0u8; 20];
                            let len = info_hash.len().min(20);
                            info_hash_arr[..len].copy_from_slice(&info_hash[..len]);
                            let _ = self.info_hash_tx.send(info_hash_arr);
                        }
                        // Respond
                        let mut r = BTreeMap::new();
                        r.insert("id".to_string(), KrpcValue::Bytes(self.our_id.to_vec()));
                        let resp = KrpcMessage::Response {
                            t: t.clone(),
                            y: "r".to_string(),
                            r,
                        };
                        let _ = self.socket.send_to(&resp.to_bytes(), src).await;
                    }
                    _ => {
                        tracing::debug!("unknown query type from {}: {}", src, q);
                    }
                }
            }
            KrpcMessage::Response { r, .. } => {
                // Handle nodes in response
                if let Some(KrpcValue::Bytes(nodes_data)) = r.get("nodes") {
                    let contacts = decode_compact_nodes(nodes_data);
                    for contact in contacts {
                        self.routing.add_node(contact.id, contact.addr);
                    }
                }
                // Handle peers in get_peers response
                let (peers, nodes) = parse_get_peers_response(&msg);
                // Note: peers are just addresses (no IDs), so we publish info_hashes via
                // the query tracking map - this is handled in discover_peers
                for node in nodes {
                    self.routing.add_node(node.id, node.addr);
                }
            }
            KrpcMessage::Error { e, .. } => {
                tracing::debug!("krpc error from {}: {} ({})", src, e.description, e.code);
            }
        }
    }

    /// Periodically send get_peers queries for random info_hashes
    async fn discover_peers(&mut self) {
        let info_hash = random_node_id(); // Use random ID as info_hash target
        let closest = self.routing.closest_nodes(&info_hash, 16);

        if closest.is_empty() {
            // Re-bootstrap if we have no nodes
            self.bootstrap().await;
            return;
        }

        for node in &closest {
            let msg = build_get_peers_query(&self.our_id, &info_hash, next_transaction_id());
            let data = msg.to_bytes();
            if let Err(e) = self.socket.send_to(&data, node.addr).await {
                tracing::debug!("send get_peers to {}: {}", node.addr, e);
            }
        }

        tracing::debug!(
            "discover_peers: sent to {} nodes, routing table has {} nodes",
            closest.len(),
            self.routing.node_count()
        );
    }

    pub fn node_count(&self) -> usize {
        self.routing.node_count()
    }
}
```

```rust
use std::collections::BTreeMap;

pub const BT_PROTOCOL: &[u8] = b"BitTorrent protocol";
pub const HANDSHAKE_LEN: usize = 68;
pub const PROTOCOL_HEADER_LEN: usize = 4; // 4-byte message length prefix

/// Message types (IDs)
pub const MSG_CHOKE: u8 = 0;
pub const MSG_UNCHOKE: u8 = 1;
pub const MSG_INTERESTED: u8 = 2;
pub const MSG_NOT_INTERESTED: u8 = 3;
pub const MSG_HAVE: u8 = 4;
pub const MSG_BITFIELD: u8 = 5;
pub const MSG_REQUEST: u8 = 6;
pub const MSG_PIECE: u8 = 7;
pub const MSG_CANCEL: u8 = 8;
pub const MSG_PORT: u8 = 9;
pub const MSG_EXTENDED: u8 = 20;

/// The BitTorrent handshake message.
#[derive(Debug, Clone)]
pub struct Handshake {
    pub info_hash: [u8; 20],
    pub peer_id: [u8; 20],
    pub reserved: [u8; 8],
}

impl Handshake {
    pub fn new(info_hash: [u8; 20], peer_id: [u8; 20]) -> Self {
        let mut reserved = [0u8; 8];
        // Set bit 43 (byte 5, bit 3) from LSB for DHT support
        reserved[5] |= 0x01; // Mainline DHT
        // Set bit 44 (byte 5, bit 4) for extended messaging
        reserved[5] |= 0x10; // Extension protocol

        Handshake {
            info_hash,
            peer_id,
            reserved,
        }
    }

    pub fn to_bytes(&self) -> Vec<u8> {
        let mut buf = Vec::with_capacity(HANDSHAKE_LEN);
        buf.push(BT_PROTOCOL.len() as u8);
        buf.extend_from_slice(BT_PROTOCOL);
        buf.extend_from_slice(&self.reserved);
        buf.extend_from_slice(&self.info_hash);
        buf.extend_from_slice(&self.peer_id);
        buf
    }

    pub fn from_bytes(data: &[u8]) -> Option<Self> {
        if data.len() < HANDSHAKE_LEN {
            return None;
        }
        let pstr_len = data[0] as usize;
        if pstr_len != 19 || &data[1..20] != BT_PROTOCOL {
            return None;
        }

        let mut reserved = [0u8; 8];
        reserved.copy_from_slice(&data[20..28]);

        let mut info_hash = [0u8; 20];
        info_hash.copy_from_slice(&data[28..48]);

        let mut peer_id = [0u8; 20];
        peer_id.copy_from_slice(&data[48..68]);

        Some(Handshake {
            info_hash,
            peer_id,
            reserved,
        })
    }

    /// Check if the peer supports extended messaging (BEP 10).
    pub fn supports_extensions(&self) -> bool {
        self.reserved[5] & 0x10 != 0
    }
}

/// A wire protocol message.
#[derive(Debug, Clone)]
pub enum WireMessage {
    KeepAlive,
    Choke,
    Unchoke,
    Interested,
    NotInterested,
    Have(u32),
    Bitfield(Vec<u8>),
    Request { index: u32, begin: u32, length: u32 },
    Piece { index: u32, begin: u32, block: Vec<u8> },
    Cancel { index: u32, begin: u32, length: u32 },
    Port(u16),
    Extended { id: u8, payload: Vec<u8> },
}

impl WireMessage {
    pub fn to_bytes(&self) -> Vec<u8> {
        let mut payload = Vec::new();

        match self {
            WireMessage::KeepAlive => {
                // 4 zero bytes = keepalive
                return vec![0, 0, 0, 0];
            }
            WireMessage::Choke => payload.push(MSG_CHOKE),
            WireMessage::Unchoke => payload.push(MSG_UNCHOKE),
            WireMessage::Interested => payload.push(MSG_INTERESTED),
            WireMessage::NotInterested => payload.push(MSG_NOT_INTERESTED),
            WireMessage::Have(piece) => {
                payload.push(MSG_HAVE);
                payload.extend_from_slice(&piece.to_be_bytes());
            }
            WireMessage::Bitfield(data) => {
                payload.push(MSG_BITFIELD);
                payload.extend_from_slice(data);
            }
            WireMessage::Request { index, begin, length } => {
                payload.push(MSG_REQUEST);
                payload.extend_from_slice(&index.to_be_bytes());
                payload.extend_from_slice(&begin.to_be_bytes());
                payload.extend_from_slice(&length.to_be_bytes());
            }
            WireMessage::Piece { index, begin, block } => {
                payload.push(MSG_PIECE);
                payload.extend_from_slice(&index.to_be_bytes());
                payload.extend_from_slice(&begin.to_be_bytes());
                payload.extend_from_slice(block);
            }
            WireMessage::Cancel { index, begin, length } => {
                payload.push(MSG_CANCEL);
                payload.extend_from_slice(&index.to_be_bytes());
                payload.extend_from_slice(&begin.to_be_bytes());
                payload.extend_from_slice(&length.to_be_bytes());
            }
            WireMessage::Port(port) => {
                payload.push(MSG_PORT);
                payload.extend_from_slice(&port.to_be_bytes());
            }
            WireMessage::Extended { id, payload: ext_payload } => {
                payload.push(MSG_EXTENDED);
                payload.push(*id);
                payload.extend_from_slice(ext_payload);
            }
        }

        let mut msg = Vec::with_capacity(4 + payload.len());
        msg.extend_from_slice(&(payload.len() as u32).to_be_bytes());
        msg.extend_from_slice(&payload);
        msg
    }

    pub fn from_bytes(data: &[u8]) -> Option<Self> {
        if data.len() < 4 {
            return None;
        }
        let mut len_bytes = [0u8; 4];
        len_bytes.copy_from_slice(&data[0..4]);
        let len = u32::from_be_bytes(len_bytes) as usize;

        if len == 0 {
            return Some(WireMessage::KeepAlive);
        }

        if data.len() < 4 + len {
            return None;
        }

        let payload = &data[4..4 + len];
        if payload.is_empty() {
            return Some(WireMessage::KeepAlive);
        }

        let msg_id = payload[0];
        let body = &payload[1..];

        match msg_id {
            MSG_CHOKE => Some(WireMessage::Choke),
            MSG_UNCHOKE => Some(WireMessage::Unchoke),
            MSG_INTERESTED => Some(WireMessage::Interested),
            MSG_NOT_INTERESTED => Some(WireMessage::NotInterested),
            MSG_HAVE if body.len() >= 4 => {
                let piece = u32::from_be_bytes([body[0], body[1], body[2], body[3]]);
                Some(WireMessage::Have(piece))
            }
            MSG_BITFIELD => Some(WireMessage::Bitfield(body.to_vec())),
            MSG_REQUEST if body.len() >= 12 => {
                let index = u32::from_be_bytes([body[0], body[1], body[2], body[3]]);
                let begin = u32::from_be_bytes([body[4], body[5], body[6], body[7]]);
                let length = u32::from_be_bytes([body[8], body[9], body[10], body[11]]);
                Some(WireMessage::Request { index, begin, length })
            }
            MSG_PIECE if body.len() >= 8 => {
                let index = u32::from_be_bytes([body[0], body[1], body[2], body[3]]);
                let begin = u32::from_be_bytes([body[4], body[5], body[6], body[7]]);
                let block = body[8..].to_vec();
                Some(WireMessage::Piece { index, begin, block })
            }
            MSG_CANCEL if body.len() >= 12 => {
                let index = u32::from_be_bytes([body[0], body[1], body[2], body[3]]);
                let begin = u32::from_be_bytes([body[4], body[5], body[6], body[7]]);
                let length = u32::from_be_bytes([body[8], body[9], body[10], body[11]]);
                Some(WireMessage::Cancel { index, begin, length })
            }
            MSG_PORT if body.len() >= 2 => {
                let port = u16::from_be_bytes([body[0], body[1]]);
                Some(WireMessage::Port(port))
            }
            MSG_EXTENDED if !body.is_empty() => {
                let ext_id = body[0];
                let ext_payload = body[1..].to_vec();
                Some(WireMessage::Extended {
                    id: ext_id,
                    payload: ext_payload,
                })
            }
            _ => None,
        }
    }
}

/// Extended handshake message (BEP 10).
#[derive(Debug)]
pub struct ExtendedHandshake {
    pub m: BTreeMap<String, u32>,
    pub metadata_size: Option<u32>,
    pub v: Option<String>,
    pub your_ip: Option<String>,
    pub reqq: Option<u32>,
}

impl ExtendedHandshake {
    pub fn to_dict(&self) -> BTreeMap<Vec<u8>, serde_bencode::value::Value> {
        let mut dict = BTreeMap::new();

        // m dict
        let mut m_dict = BTreeMap::new();
        for (k, v) in &self.m {
            m_dict.insert(k.as_bytes().to_vec(), serde_bencode::value::Value::Int(*v as i64));
        }
        dict.insert(b"m".to_vec(), serde_bencode::value::Value::Dict(m_dict));

        if let Some(size) = self.metadata_size {
            dict.insert(b"metadata_size".to_vec(), serde_bencode::value::Value::Int(size as i64));
        }

        if let Some(ref v) = self.v {
            dict.insert(b"v".to_vec(), serde_bencode::value::Value::Bytes(v.as_bytes().to_vec()));
        }

        if let Some(ref ip) = self.your_ip {
            dict.insert(b"yourip".to_vec(), serde_bencode::value::Value::Bytes(ip.as_bytes().to_vec()));
        }

        if let Some(reqq) = self.reqq {
            dict.insert(b"reqq".to_vec(), serde_bencode::value::Value::Int(reqq as i64));
        }

        dict
    }

    pub fn from_bytes(data: &[u8]) -> Option<Self> {
        let value: serde_bencode::value::Value = serde_bencode::from_bytes(data).ok()?;

        let dict = match &value {
            serde_bencode::value::Value::Dict(d) => d,
            _ => return None,
        };

        let mut m = BTreeMap::new();
        if let Some(serde_bencode::value::Value::Dict(m_dict)) =
            dict.get(&b"m".to_vec())
        {
            for (k, v) in m_dict {
                if let serde_bencode::value::Value::Int(i) = v {
                    if let Ok(key) = String::from_utf8(k.clone()) {
                        m.insert(key, *i as u32);
                    }
                }
            }
        }

        let metadata_size = match dict.get(&b"metadata_size".to_vec()) {
            Some(serde_bencode::value::Value::Int(i)) => Some(*i as u32),
            _ => None,
        };

        let v = match dict.get(&b"v".to_vec()) {
            Some(serde_bencode::value::Value::Bytes(b)) => {
                Some(String::from_utf8_lossy(b).to_string())
            }
            _ => None,
        };

        let your_ip = match dict.get(&b"yourip".to_vec()) {
            Some(serde_bencode::value::Value::Bytes(b)) => {
                Some(String::from_utf8_lossy(b).to_string())
            }
            _ => None,
        };

        let reqq = match dict.get(&b"reqq".to_vec()) {
            Some(serde_bencode::value::Value::Int(i)) => Some(*i as u32),
            _ => None,
        };

        Some(ExtendedHandshake {
            m,
            metadata_size,
            v,
            your_ip,
            reqq,
        })
    }
}

/// ut_metadata message types (BEP 9).
pub const UT_METADATA_REQUEST: u8 = 0;
pub const UT_METADATA_DATA: u8 = 1;

/// Build a ut_metadata request for a specific piece.
pub fn build_metadata_request(piece: u32) -> Vec<u8> {
    let mut dict = BTreeMap::new();
    dict.insert(
        b"msg_type".to_vec(),
        serde_bencode::value::Value::Int(UT_METADATA_REQUEST as i64),
    );
    dict.insert(
        b"piece".to_vec(),
        serde_bencode::value::Value::Int(piece as i64),
    );
    serde_bencode::to_bytes(&serde_bencode::value::Value::Dict(dict)).unwrap_or_default()
}

/// Parse a ut_metadata response. Returns (piece_index, total_size, data).
pub fn parse_metadata_response(data: &[u8]) -> Option<(u32, u32, Vec<u8>)> {
    let value: serde_bencode::value::Value = serde_bencode::from_bytes(data).ok()?;
    let dict = match &value {
        serde_bencode::value::Value::Dict(d) => d,
        _ => return None,
    };

    let msg_type = match dict.get(&b"msg_type".to_vec()) {
        Some(serde_bencode::value::Value::Int(i)) => *i as u8,
        _ => return None,
    };

    if msg_type != UT_METADATA_DATA {
        return None;
    }

    let piece = match dict.get(&b"piece".to_vec()) {
        Some(serde_bencode::value::Value::Int(i)) => *i as u32,
        _ => return None,
    };

    let total_size = match dict.get(&b"total_size".to_vec()) {
        Some(serde_bencode::value::Value::Int(i)) => *i as u32,
        _ => return None,
    };

    // The actual metadata piece data follows the bencoded dict in the message
    // Find the end of the bencoded dict and take the rest
    let dict_end = find_bencode_end(data)?;
    let meta_data = data[dict_end..].to_vec();

    Some((piece, total_size, meta_data))
}

/// Find the end of a bencoded value in bytes. Returns the index after the value.
fn find_bencode_end(data: &[u8]) -> Option<usize> {
    let mut pos = 0;
    fn parse_next(data: &[u8], pos: &mut usize) {
        if *pos >= data.len() {
            return;
        }
        match data[*pos] {
            b'd' => {
                *pos += 1;
                while *pos < data.len() && data[*pos] != b'e' {
                    // key (always a string)
                    parse_next(data, pos);
                    if *pos >= data.len() {
                        return;
                    }
                    // value
                    parse_next(data, pos);
                }
                *pos += 1; // skip 'e'
            }
            b'l' => {
                *pos += 1;
                while *pos < data.len() && data[*pos] != b'e' {
                    parse_next(data, pos);
                }
                *pos += 1; // skip 'e'
            }
            b'i' => {
                *pos += 1;
                while *pos < data.len() && data[*pos] != b'e' {
                    *pos += 1;
                }
                *pos += 1; // skip 'e'
            }
            _ => {
                // string: <length>:<bytes>
                let mut len_end = *pos;
                while len_end < data.len() && data[len_end].is_ascii_digit() {
                    len_end += 1;
                }
                if len_end < data.len() && data[len_end] == b':' {
                    let len_str = std::str::from_utf8(&data[*pos..len_end]).ok();
                    let byte_len: usize = len_str.and_then(|s| s.parse().ok()).unwrap_or(0);
                    *pos = len_end + 1 + byte_len;
                }
            }
        }
    }
    parse_next(data, &mut pos);
    Some(pos)
}
```

```rust
/// Fetch metadata for an info_hash from a specific peer.

pub async fn fetch_from_peer(
    info_hash: &InfoHash,
    peer: &PeerContact,
) -> Result<Vec<u8>, String> {
    let addr = std::net::SocketAddr::V4(peer.addr);
    let stream = timeout(Duration::from_secs(30), TcpStream::connect(addr))
        .await
        .map_err(|_| "connect timeout".to_string())?
        .map_err(|e| format!("connect error: {}", e))?;

    fetch_from_stream(info_hash, stream).await
}

async fn fetch_from_stream(
    info_hash: &InfoHash,
    mut stream: TcpStream,
) -> Result<Vec<u8>, String> {
    let our_peer_id = crate::types::random_node_id();

    // Send handshake
    let hs = Handshake::new(*info_hash, our_peer_id);
    stream
        .write_all(&hs.to_bytes())
        .await
        .map_err(|e| format!("write handshake: {}", e))?;

    // Read handshake
    let mut buf = vec![0u8; 1024];
    let n = timeout(Duration::from_secs(30), stream.read(&mut buf))
        .await
        .map_err(|_| "handshake read timeout".to_string())?
        .map_err(|e| format!("read error: {}", e))?;

    let hs_resp = Handshake::from_bytes(&buf[..n])
        .ok_or_else(|| "invalid handshake response".to_string())?;

    if !hs_resp.supports_extensions() {
        return Err("peer doesn't support extensions".to_string());
    }

    // Send extended handshake
    let mut m = std::collections::BTreeMap::new();
    m.insert("ut_metadata".to_string(), 1u32);

    let ext_hs = ExtendedHandshake {
        m,
        metadata_size: None,
        v: Some("btfind 0.1".to_string()),
        your_ip: None,
        reqq: None,
    };

    let ext_hs_bytes = serde_bencode::to_bytes(
        &serde_bencode::value::Value::Dict(ext_hs.to_dict()),
    )
    .map_err(|e| format!("encode ext hs: {}", e))?;

    let ext_msg = WireMessage::Extended {
        id: 0,
        payload: ext_hs_bytes,
    };

    stream
        .write_all(&ext_msg.to_bytes())
        .await
        .map_err(|e| format!("write ext hs: {}", e))?;

    // Read extended handshake response
    let mut buf = vec![0u8; 65536];
    let n = timeout(Duration::from_secs(30), stream.read(&mut buf))
        .await
        .map_err(|_| "ext hs read timeout".to_string())?
        .map_err(|e| format!("read error: {}", e))?;

    let msg = WireMessage::from_bytes(&buf[..n])
        .ok_or_else(|| "invalid wire message".to_string())?;

    let (ext_id, ext_payload) = match msg {
        WireMessage::Extended { id, payload } if id == 0 => (id, payload),
        _ => return Err("expected extended handshake response".to_string()),
    };

    let peer_ext_hs = ExtendedHandshake::from_bytes(&ext_payload)
        .ok_or_else(|| "invalid extended handshake".to_string())?;

    // Get ut_metadata message ID
    let ut_metadata_id = peer_ext_hs
        .m
        .get("ut_metadata")
        .ok_or_else(|| "peer doesn't support ut_metadata".to_string())?;

    let metadata_size = peer_ext_hs
        .metadata_size
        .ok_or_else(|| "peer didn't advertise metadata_size".to_string())?;

    let total_pieces = metadata_piece_count(metadata_size);
    let mut pieces: Vec<Option<Vec<u8>>> = vec![None; total_pieces as usize];
    let mut received = 0u32;

    // Request all pieces
    for piece in 0..total_pieces {
        let req = wire::build_metadata_request(piece);
        let req_msg = WireMessage::Extended {
            id: *ut_metadata_id as u8,
            payload: req,
        };
        stream
            .write_all(&req_msg.to_bytes())
            .await
            .map_err(|e| format!("write metadata request: {}", e))?;

        tracing::debug!("requested metadata piece {}/{}", piece + 1, total_pieces);
    }

    // Read responses
    let mut buf = vec![0u8; 65536];
    let deadline = Duration::from_secs(60);
    let start = std::time::Instant::now();

    while received < total_pieces {
        let remaining = deadline.saturating_sub(start.elapsed());
        if remaining.is_zero() {
            return Err("metadata receive timeout".to_string());
        }

        let n = timeout(remaining, stream.read(&mut buf))
            .await
            .map_err(|_| "receive timeout".to_string())?
            .map_err(|e| format!("read error: {}", e))?;

        if n == 0 {
            break;
        }

        let msg = WireMessage::from_bytes(&buf[..n]);
        match msg {
            Some(WireMessage::Extended { id, payload }) if id == *ut_metadata_id as u8 => {
                if let Some((piece_idx, _total_size, data)) =
                    wire::parse_metadata_response(&payload)
                {
                    if (piece_idx as usize) < pieces.len() && pieces[piece_idx as usize].is_none() {
                        pieces[piece_idx as usize] = Some(data);
                        received += 1;
                        tracing::debug!(
                            "received metadata piece {}/{}",
                            piece_idx + 1,
                            total_pieces
                        );
                    }
                }
            }
            Some(WireMessage::KeepAlive) => {
                // Ok, just keep waiting
            }
            _ => {
                // Other messages; ignore or respond with interested/unchoke
                tracing::debug!("ignoring non-metadata message");
            }
        }
    }

    // Assemble all pieces
    let mut all_data = Vec::with_capacity(metadata_size as usize);
    for piece in &pieces {
        match piece {
            Some(data) => all_data.extend_from_slice(data),
            None => return Err("missing metadata piece".to_string()),
        }
    }

    if all_data.is_empty() {
        return Err("empty metadata".to_string());
    }

    Ok(all_data)
}

/// Parse torrent info from bencoded metadata.
pub fn parse_torrent_info(data: &[u8]) -> Result<TorrentInfo, String> {
    let value: serde_bencode::value::Value =
        serde_bencode::from_bytes(data).map_err(|e| format!("bencode error: {}", e))?;

    let dict = match value {
        serde_bencode::value::Value::Dict(d) => d,
        _ => return Err("not a dict".to_string()),
    };

    let name = get_bencode_str(&dict, "name").unwrap_or_else(|| "unknown".to_string());
    let piece_length = get_bencode_int(&dict, "piece length").unwrap_or(0);

    let (total_size, files) = if let Some(serde_bencode::value::Value::List(file_list)) =
        dict.get(&b"files".to_vec())
    {
        // Multi-file torrent
        let mut files_out = Vec::new();
        let mut total = 0i64;
        for file_val in file_list {
            if let serde_bencode::value::Value::Dict(file_dict) = file_val {
                let file_size = get_bencode_int(file_dict, "length").unwrap_or(0);
                total += file_size;

                let mut path_parts = Vec::new();
                if let Some(serde_bencode::value::Value::List(parts)) =
                    file_dict.get(&b"path".to_vec())
                {
                    for part in parts {
                        if let serde_bencode::value::Value::Bytes(b) = part {
                            path_parts.push(String::from_utf8_lossy(b).to_string());
                        }
                    }
                }
                let file_path = path_parts.join("/");
                files_out.push((file_path, file_size));
            }
        }
        (total, files_out)
    } else if let Some(length) = get_bencode_int(&dict, "length") {
        // Single-file torrent
        (length, vec![(name.clone(), length)])
    } else {
        return Err("no files or length in info dict".to_string());
    };

    Ok(TorrentInfo {
        name,
        piece_length,
        total_size,
        file_count: files.len() as i64,
        files,
    })
}

#[derive(Debug)]
pub struct TorrentInfo {
    pub name: String,
    pub piece_length: i64,
    pub total_size: i64,
    pub file_count: i64,
    pub files: Vec<(String, i64)>,
}

fn get_bencode_str(dict: &std::collections::BTreeMap<Vec<u8>, serde_bencode::value::Value>, key: &str) -> Option<String> {
    match dict.get(&key.as_bytes().to_vec())? {
        serde_bencode::value::Value::Bytes(b) => Some(String::from_utf8_lossy(b).to_string()),
        _ => None,
    }
}

fn get_bencode_int(dict: &std::collections::BTreeMap<Vec<u8>, serde_bencode::value::Value>, key: &str) -> Option<i64> {
    match dict.get(&key.as_bytes().to_vec())? {
        serde_bencode::value::Value::Int(i) => Some(*i),
        _ => None,
    }
}

/// Run the metadata fetcher: consumes info_hashes, fetches metadata, stores results.
pub async fn run_metadata_fetcher(
    mut info_hash_rx: mpsc::UnboundedReceiver<InfoHash>,
    store: Arc<Store>,
    max_concurrent: usize,
) {
    let semaphore = Arc::new(Semaphore::new(max_concurrent));

    while let Some(info_hash) = info_hash_rx.recv().await {
        let store = store.clone();
        let permit = semaphore.clone();

        tokio::spawn(async move {
            let _permit = permit.acquire_owned().await;

            // Check if we already have this
            if let Ok(Some(record)) = store.get_torrent(&info_hash) {
                if record.metadata_complete {
                    tracing::debug!("already have metadata for {}", hex::encode(&info_hash[..4]));
                    return;
                }
                // Check if recently attempted (within 24h)
                let now = chrono::Utc::now().timestamp();
                if now - record.last_seen < 86400 && record.name.is_none() {
                    tracing::debug!("skipping recently-attempted {}", hex::encode(&info_hash[..4]));
                    return;
                }
            }

            // Try up to 8 known peers from a get_peers query
            // For now, we need the DHT crawler to share its peer lists.
            // This is a simplification: we attempt a direct connect to a random peer
            // from the peer list associated with this info_hash.
            //
            // In a full implementation, we'd maintain a peer cache from DHT responses.
            // For this MVP, we log the info_hash and store it for later.

            let _ = store.upsert_torrent(info_hash, None, 0, 0, "dht");

            // Try to fetch from known torrents by testing connectivity
            // In practice, peers would come from the DHT's get_peers responses.
            // For now, this is a placeholder for the full peer-based fetch.

            tracing::info!("saw info_hash: {}", hex::encode(&info_hash));
        });
    }
}
```

```rust
async fn cmd_run(
    config: &config::Config,
    duration: Option<u64>,
    continuous: bool,
    max_concurrent: usize,
    port: u16,
) {
    let db_path = config.database_path();
    if let Some(parent) = db_path.parent() {
        std::fs::create_dir_all(parent).ok();
    }
    let store = Arc::new(store::Store::open(&db_path).expect("failed to open database"));

    // Bind UDP socket
    let addr: SocketAddrV4 = format!("0.0.0.0:{}", port).parse().unwrap();
    let socket = Arc::new(
        UdpSocket::bind(addr)
            .await
            .expect("failed to bind UDP socket"),
    );

    tracing::info!("bound to {}", addr);

    // Channel for info_hash discovery
    let (info_hash_tx, info_hash_rx) = mpsc::unbounded_channel();

    // Start DHT crawler
    let crawler = dht::DhtCrawler::new(socket.clone(), info_hash_tx);
    let crawler_handle = tokio::spawn(async move {
        crawler
            .run(
                Duration::from_secs(config.crawl.get_peers_interval_secs),
                Duration::from_secs(config.crawl.bucket_refresh_mins * 60),
                continuous || duration.is_none(),
            )
            .await;
    });

    // Start metadata fetcher
    let store_clone = store.clone();
    let fetcher_handle = tokio::spawn(async move {
        metadata::run_metadata_fetcher(info_hash_rx, store_clone, max_concurrent).await;
    });

    tracing::info!("crawler started");

    if let Some(secs) = duration {
        tracing::info!("running for {} seconds", secs);
        tokio::time::sleep(Duration::from_secs(secs)).await;
        tracing::info!("duration reached, shutting down");
        std::process::exit(0);
    }

    // Wait for crawler (runs forever in continuous mode)
    tokio::select! {
        _ = crawler_handle => {}
        _ = fetcher_handle => {}
        _ = tokio::signal::ctrl_c() => {
            tracing::info!("received Ctrl+C, shutting down");
            std::process::exit(0);
        }
    }
}

fn cmd_search(config: &config::Config, query: Option<String>, limit: usize, sort: &str, json: bool) {
    let db_path = config.database_path();
    let store = store::Store::open(&db_path).expect("failed to open database");

    let results = if let Some(q) = &query {
        store
            .search_torrents(q, limit)
            .expect("search failed")
    } else {
        // List recent torrents when no query
        store
            .search_torrents("", limit)
            .expect("search failed")
    };

    if json {
        let output: Vec<serde_json::Value> = results
            .iter()
            .map(|t| {
                serde_json::json!({
                    "info_hash": hex::encode(t.info_hash),
                    "name": t.name,
                    "total_size": t.total_size,
                    "file_count": t.file_count,
                    "first_seen": t.first_seen,
                    "last_seen": t.last_seen,
                })
            })
            .collect();
        println!("{}", serde_json::to_string_pretty(&output).unwrap());
    } else {
        for t in &results {
            let hash_preview = hex::encode(&t.info_hash[..4]);
            let name = t.name.as_deref().unwrap_or("<unknown>");
            let size_str = format_size(t.total_size);
            println!("{hash_preview}..  {size_str:>10}  {name}");
        }
        println!("--- {} results ---", results.len());
    }
}

fn format_size(bytes: i64) -> String {
    if bytes < 0 {
        return "0 B".to_string();
    }
    let bytes = bytes as u64;
    const UNITS: &[&str] = &["B", "KB", "MB", "GB", "TB"];
    let mut size = bytes as f64;
    let mut unit_idx = 0;
    while size >= 1024.0 && unit_idx < UNITS.len() - 1 {
        size /= 1024.0;
        unit_idx += 1;
    }
    if unit_idx == 0 {
        format!("{} {}", bytes, UNITS[unit_idx])
    } else {
        format!("{:.1} {}", size, UNITS[unit_idx])
    }
}
```
